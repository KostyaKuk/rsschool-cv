Code example

const bubbleSort = (arr) => {
  const a = [...arr];
  for (let i = 0; i < a.length - 1; i++) {
    for (let j = 0; j < a.length - 1 - i; j++) {
      if (a[j] > a[j + 1]) {
        [a[j], a[j + 1]] = [a[j + 1], a[j]];
      }
    }
  }
  return a;
};

console.log(bubbleSort([5, 3, 8, 1, 9, 2]));
// [1, 2, 3, 5, 8, 9]
